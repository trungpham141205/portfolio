module alu_4bit (
    input  [3:0] a,
    input  [3:0] b,
    input  [1:0] alu_sel,
    output reg [3:0] result,
    output reg       carry_out
);
    wire [4:0] add_res;
    assign add_res = a + b;

    always @(*) begin
        result    = 4'b0000;
        carry_out = 1'b0;
        case (alu_sel)
            2'b00: result = a & b;
            2'b01: result = a | b;
            2'b10: result = a ^ b;
            2'b11: begin
                result    = add_res[3:0];
                carry_out = add_res[4];
            end
        endcase
    end
endmodule
FSM

Traffic Light Controller

Current State Register S_RED 
module traffic_fsm (
    input clk,
    input rst_n,
    output reg red, yellow, green
);
    parameter S_RED = 2'b00, S_GREEN = 2'b01, S_YELLOW = 2'b10;
    reg [1:0] state, next_state;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) state <= S_RED;
        else        state <= next_state;
    end

    always @(*) begin
        case (state)
            S_RED:    next_state = S_GREEN;
            S_GREEN:  next_state = S_YELLOW;
            S_YELLOW: next_state = S_RED;
            default:  next_state = S_RED;
        endcase
    end

    always @(*) begin
        red=0; yellow=0; green=0;
        if (state == S_RED) red = 1;
        else if (state == S_GREEN) green = 1;
        else if (state == S_YELLOW) yellow = 1;
    end
endmodule
